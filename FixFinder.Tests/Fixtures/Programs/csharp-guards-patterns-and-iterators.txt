using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Threading.Tasks;

public static class Program
{
    static int ReadAge(string? text)
    {
        if (string.IsNullOrWhiteSpace(text))
        {
            return -1;
        }

        return int.TryParse(text.Trim(), out var age) && age >= 0 ? age : -1;
    }

    static string Nickname(Dictionary<string, string> nicknames, string name) =>
        nicknames.TryGetValue(name, out var nickname) ? nickname : name;

    static int FirstEven(List<int> numbers)
    {
        if (!numbers.Any(number => number % 2 == 0))
        {
            return 0;
        }

        return numbers.First(number => number % 2 == 0);
    }

    static int Undo(Stack<int> history)
    {
        if (history.Count == 0)
        {
            return 0;
        }

        return history.Pop();
    }

    static IEnumerable<int> Squares(int count)
    {
        for (var i = 0; i < count; i++)
        {
            yield return i * i;
        }
    }

    static string Classify(object value) => value switch
    {
        int n when n > 0 => "positive",
        int => "not positive",
        string { Length: 0 } => "empty text",
        string text => $"text of {text.Length}",
        null => "nothing",
        _ => "something else",
    };

    static List<string> cache = null!;

    static List<string> Cache()
    {
        cache ??= new List<string>();
        return cache;
    }

    static async Task<int> SumAsync(IEnumerable<Func<Task<int>>> sources)
    {
        var results = await Task.WhenAll(sources.Select(source => source()));
        return results.Sum();
    }

    record Point(int X, int Y);

    public static async Task Main(string[] args)
    {
        Console.WriteLine(ReadAge(" 42 ") + ReadAge(null) + ReadAge("abc"));
        Console.WriteLine(Nickname(new Dictionary<string, string> { ["Robert"] = "Bob" }, "Robert"));
        Console.WriteLine(FirstEven(new List<int> { 1, 3, 4 }) + FirstEven(new List<int>()));

        var history = new Stack<int>();
        history.Push(3);
        Console.WriteLine(Undo(history) + Undo(history));

        Console.WriteLine(string.Join(",", Squares(4)));
        Console.WriteLine(Classify(5) + Classify("") + Classify(null!));
        Cache().Add("x");
        Console.WriteLine(Cache().Count);

        var moved = new Point(1, 2) with { Y = 5 };
        Console.WriteLine(moved);

        using var writer = new StringWriter();
        writer.Write("kept");
        Console.WriteLine(writer.ToString());

        Console.WriteLine(await SumAsync(new Func<Task<int>>[] { () => Task.FromResult(1), () => Task.FromResult(2) }));
        if (args.Length > 0)
        {
            Console.WriteLine(args[0].ToUpperInvariant());
        }
    }
}
