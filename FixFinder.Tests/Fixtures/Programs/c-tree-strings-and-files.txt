#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

typedef struct Node {
    int key;
    struct Node *left;
    struct Node *right;
} Node;

static Node *insert(Node *root, int key)
{
    if (root == NULL) {
        Node *made = malloc(sizeof(Node));
        if (made == NULL) {
            return NULL;
        }
        made->key = key;
        made->left = NULL;
        made->right = NULL;
        return made;
    }
    if (key < root->key) {
        root->left = insert(root->left, key);
    } else if (key > root->key) {
        root->right = insert(root->right, key);
    }
    return root;
}

static void print_in_order(const Node *root)
{
    if (root == NULL) {
        return;
    }
    print_in_order(root->left);
    printf("%d ", root->key);
    print_in_order(root->right);
}

static void free_tree(Node *root)
{
    if (root == NULL) {
        return;
    }
    free_tree(root->left);
    free_tree(root->right);
    free(root);
}

static void reverse(char *text)
{
    size_t length = strlen(text);
    if (length < 2) {
        return;
    }
    for (size_t i = 0, j = length - 1; i < j; i++, j--) {
        char kept = text[i];
        text[i] = text[j];
        text[j] = kept;
    }
}

static int count_words(const char *text)
{
    int words = 0;
    int inside = 0;
    for (const char *c = text; *c != '\0'; c++) {
        if (isspace((unsigned char)*c)) {
            inside = 0;
        } else if (!inside) {
            inside = 1;
            words++;
        }
    }
    return words;
}

static int sum_file(const char *path, long *total)
{
    FILE *file = fopen(path, "r");
    if (file == NULL) {
        return -1;
    }

    char line[128];
    *total = 0;
    while (fgets(line, sizeof line, file) != NULL) {
        line[strcspn(line, "\n")] = '\0';
        if (line[0] == '\0') {
            continue;
        }
        *total += strtol(line, NULL, 10);
    }

    fclose(file);
    return 0;
}

int main(int argc, char *argv[])
{
    Node *root = NULL;
    int keys[] = {5, 3, 8, 1, 4};
    for (size_t i = 0; i < sizeof keys / sizeof keys[0]; i++) {
        Node *grown = insert(root, keys[i]);
        if (grown == NULL) {
            free_tree(root);
            return 1;
        }
        root = grown;
    }
    print_in_order(root);
    printf("\n");
    free_tree(root);

    char word[32];
    snprintf(word, sizeof word, "%s", "stressed");
    reverse(word);
    printf("%s %d\n", word, count_words("  two   words "));

    if (argc > 1) {
        long total;
        if (sum_file(argv[1], &total) == 0) {
            printf("%ld\n", total);
        } else {
            perror(argv[1]);
        }
    }
    return 0;
}
