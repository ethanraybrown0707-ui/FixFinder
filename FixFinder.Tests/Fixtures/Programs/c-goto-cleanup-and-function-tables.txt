#include <errno.h>
#include <limits.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

enum Colour { RED, GREEN, BLUE, COLOURS };

static const char *const names[COLOURS] = {"red", "green", "blue"};

typedef int (*Operation)(int, int);

static int add(int a, int b) { return a + b; }
static int multiply(int a, int b) { return a * b; }

static const Operation operations[] = {add, multiply};

static int parse_number(const char *text, long *out)
{
    char *end;
    errno = 0;
    long value = strtol(text, &end, 10);
    if (errno != 0 || end == text || *end != '\0') {
        return -1;
    }
    if (value < INT_MIN || value > INT_MAX) {
        return -1;
    }
    *out = value;
    return 0;
}

static int copy_file(const char *from, const char *to)
{
    int status = -1;
    char *buffer = NULL;
    FILE *in = NULL;
    FILE *out = NULL;

    in = fopen(from, "rb");
    if (in == NULL) {
        goto cleanup;
    }
    out = fopen(to, "wb");
    if (out == NULL) {
        goto cleanup;
    }
    buffer = malloc(4096);
    if (buffer == NULL) {
        goto cleanup;
    }

    size_t read;
    while ((read = fread(buffer, 1, 4096, in)) > 0) {
        if (fwrite(buffer, 1, read, out) != read) {
            goto cleanup;
        }
    }
    status = 0;

cleanup:
    free(buffer);
    if (out != NULL) {
        fclose(out);
    }
    if (in != NULL) {
        fclose(in);
    }
    return status;
}

static int hanoi(int disks, int from, int to, int spare)
{
    if (disks == 0) {
        return 0;
    }
    return hanoi(disks - 1, from, spare, to) + 1 + hanoi(disks - 1, spare, to, from);
}

typedef struct {
    unsigned ready : 1;
    unsigned level : 3;
} Flags;

typedef union {
    int whole;
    unsigned char bytes[sizeof(int)];
} View;

int main(int argc, char *argv[])
{
    long number = 0;
    if (argc > 1 && parse_number(argv[1], &number) == 0) {
        printf("%ld\n", number);
    }

    for (int colour = RED; colour < COLOURS; colour++) {
        printf("%s\n", names[colour]);
    }

    for (size_t i = 0; i < sizeof operations / sizeof operations[0]; i++) {
        printf("%d\n", operations[i](3, 4));
    }

    char source[] = "copy me";
    char target[sizeof source];
    memcpy(target, source, sizeof source);
    memmove(target + 1, target, 3);
    printf("%s %d\n", target, hanoi(3, 1, 3, 2));

    Flags flags = {1, 5};
    View view;
    view.whole = 1;
    printf("%u %u %u\n", flags.ready, flags.level, (unsigned)view.bytes[0]);

    if (argc > 2 && copy_file(argv[1], argv[2]) != 0) {
        perror("copy");
        return 1;
    }
    return 0;
}
