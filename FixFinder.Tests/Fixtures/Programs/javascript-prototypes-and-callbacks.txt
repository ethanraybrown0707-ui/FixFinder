var Counter = (function () {
  var created = 0;

  function Counter(start) {
    if (!(this instanceof Counter)) {
      return new Counter(start);
    }
    this.value = start || 0;
    created += 1;
  }

  Counter.prototype.increment = function (by) {
    this.value += by === undefined ? 1 : by;
    return this;
  };

  Counter.prototype.toString = function () {
    return 'Counter(' + this.value + ')';
  };

  Counter.created = function () {
    return created;
  };

  return Counter;
})();

function readConfig(text, callback) {
  var parsed;
  try {
    parsed = JSON.parse(text);
  } catch (error) {
    callback(error, null);
    return;
  }
  callback(null, parsed);
}

function sum() {
  var total = 0;
  for (var i = 0; i < arguments.length; i++) {
    total += arguments[i];
  }
  return total;
}
