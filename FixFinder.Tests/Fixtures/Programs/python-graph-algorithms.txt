from graphs import Graph, UnionFind


def main():
    roads = Graph()
    for start, end, miles in [("a", "b", 4), ("a", "c", 2), ("c", "b", 1), ("b", "d", 5)]:
        roads.connect(start, end, miles)

    distances = roads.shortest_from("a")
    for town in sorted(distances):
        print(town, distances[town])

    print(roads.reachable_from("a"))
    print(roads.order(["shop", "cook", "eat"], [("shop", "cook"), ("cook", "eat")]))

    groups = UnionFind(5)
    groups.join(0, 1)
    groups.join(3, 4)
    print(groups.same(0, 1), groups.same(1, 3), groups.count)

import heapq
from collections import deque


class Graph:
    def __init__(self):
        self.edges = {}

    def connect(self, start, end, weight):
        self.edges.setdefault(start, []).append((end, weight))
        self.edges.setdefault(end, []).append((start, weight))

    def shortest_from(self, source):
        distances = {source: 0}
        queue = [(0, source)]
        while queue:
            distance, town = heapq.heappop(queue)
            if distance > distances.get(town, float("inf")):
                continue
            for neighbour, weight in self.edges.get(town, []):
                candidate = distance + weight
                if candidate < distances.get(neighbour, float("inf")):
                    distances[neighbour] = candidate
                    heapq.heappush(queue, (candidate, neighbour))
        return distances

    def reachable_from(self, source):
        seen = {source}
        waiting = deque([source])
        order = []
        while waiting:
            town = waiting.popleft()
            order.append(town)
            for neighbour, _ in self.edges.get(town, []):
                if neighbour not in seen:
                    seen.add(neighbour)
                    waiting.append(neighbour)
        return order

    @staticmethod
    def order(tasks, before):
        incoming = {task: 0 for task in tasks}
        after = {task: [] for task in tasks}
        for first, second in before:
            after[first].append(second)
            incoming[second] += 1
        ready = deque(task for task in tasks if incoming[task] == 0)
        ordered = []
        while ready:
            task = ready.popleft()
            ordered.append(task)
            for follower in after[task]:
                incoming[follower] -= 1
                if incoming[follower] == 0:
                    ready.append(follower)
        if len(ordered) != len(tasks):
            raise ValueError("the tasks wait on each other in a circle")
        return ordered


class UnionFind:
    def __init__(self, size):
        self.parent = list(range(size))
        self.rank = [0] * size
        self.count = size

    def find(self, item):
        while self.parent[item] != item:
            self.parent[item] = self.parent[self.parent[item]]
            item = self.parent[item]
        return item

    def join(self, first, second):
        root_first, root_second = self.find(first), self.find(second)
        if root_first == root_second:
            return False
        if self.rank[root_first] < self.rank[root_second]:
            root_first, root_second = root_second, root_first
        self.parent[root_second] = root_first
        if self.rank[root_first] == self.rank[root_second]:
            self.rank[root_first] += 1
        self.count -= 1
        return True

    def same(self, first, second):
        return self.find(first) == self.find(second)
